PRINCIPLE 01

Networks lie.

Packets drop, reorder, and duplicate. Clocks drift. Servers die mid-call. Every distributed system is a defence against these four facts.

PRINCIPLE 02

Partial failure is default.

In a single process things work or crash. In a distributed system, half the operation succeeds and the other half doesn't reply.

PRINCIPLE 03

There is no now.

Two events on different machines have no objective order. You either accept this, or invent ordering with logical clocks.

PRINCIPLE 04

Trade-offs, not solutions.

Every distributed property comes at the cost of another. CAP, PACELC, FLP — these aren't trivia, they're the menu.

01

The Eight Fallacies of Distributed Computing

Classic Mental model
Difficulty

The story

Every junior engineer assumes the network is reliable, fast, and secure. Every senior engineer was once a junior engineer who got paged at 3am because of one of these. Peter Deutsch wrote them down at Sun in 1994 and the list hasn't aged a day.

The reason this matters: the patterns in the rest of this guide — retries, circuit breakers, timeouts, idempotency — are direct responses to these fallacies. Internalise these eight and the rest stops feeling arbitrary.

Use when
  • Designing any service-to-service call
  • Reviewing an architecture diagram
  • You catch yourself writing code without a timeout
Avoid when
  • You really are inside one process
  • Latency genuinely doesn't matter (rare)
  • You enjoy 3am pages
Connects to Circuit Breaker, Retry with Backoff, Idempotency

func callOrders(ctx context.Context, cart Cart) (*Result, error) {

    // #2 Latency > 0           → ALWAYS set a timeout

    defer cancel()

        return nil

    // #5 Topology shifts       → resolve via discovery, not hostnames
    // #6 Multiple admins       → log who/what/when on every call
    return res, err
}

import pRetry, { AbortError } from "p-retry";
import { Agent } from "undici";

export class OrdersClient {
    // #2 latency > 0  → connect timeout here, read timeout per request
    // #4 not secure  → TLS via mesh / https with the default CA store

    async place(cart: Cart): Promise<OrderResult> {
        return pRetry(async () => {                   // #1 retry transients
            const res = await fetch("https://orders/orders", {   // #5 logical name
                method: "POST",
                headers: { "content-type": "application/json" },
                body: JSON.stringify(cart),
                dispatcher: this.dispatcher,
                signal: AbortSignal.timeout(2000),
            });
            if (res.status >= 500) throw new TransientError(res.status);
            if (!res.ok) throw new AbortError(`orders: ${res.status}`);  // 4xx: no retry
            return (await res.json()) as OrderResult;
        }, {
            retries: 2,                               // 3 attempts in total
            minTimeout: 100, factor: 2, randomize: true,
        });
    }
}
02

CAP Theorem

Core Trade-off
Difficulty

The story

Eric Brewer's theorem, proved by Gilbert & Lynch: in the presence of a network partition, a distributed system can guarantee either Consistency (every read sees the latest write) or Availability (every request gets a response), but not both. Partition tolerance isn't really a choice — partitions happen — so real systems land in two camps: CP (refuse the request) or AP (return possibly-stale data).

The trick most people miss: CAP only tells you about behavior during a partition. When the network is healthy, you can have all three. The famous mistake is reading "pick two" and applying it always — it applies only when packets are dropping.

Use when
  • Choosing a primary datastore
  • Justifying read/write tradeoffs in design review
  • Explaining why etcd refuses writes when split
Avoid when
  • Citing it for "regular" performance trade-offs (use PACELC)
  • You haven't drawn what a partition looks like
  • You think every system is purely CP or purely AP
Connects to PACELC, Quorums, Raft
// CP system: refuse writes when leadership is uncertain (etcd / Raft).
func (n *Node) Write(k, v string) error {
    if !n.isLeader() {
        return ErrNotLeader             // reject — preserve C
    }
    acks := n.replicate(k, v, timeout)
    if acks < n.quorum() {
        return ErrNoQuorum             // no answer beats wrong answer
    }
    return nil
}

// AP system: accept the write locally, reconcile later (Cassandra / Dynamo).
func (n *APNode) Write(k, v string) error {
    n.local.put(k, VersionedValue{V: v, Vec: n.tick()})
    go n.gossip(k)              // best-effort propagation
    return nil                    // always say yes
}
# CP vs AP, expressed as a per-replica policy decision.
from enum import Enum

class Mode(Enum):
    CP = "cp"     # refuse during partition
    AP = "ap"     # accept and reconcile later

class Replica:
    def __init__(self, mode: Mode):
        self.mode = mode

    async def write(self, k: str, v: str) -> bool:
        peers = await self.probe_peers()
        have_quorum = len(peers) >= self.quorum_size

        if self.mode == Mode.CP:
            if not have_quorum:
                raise UnavailableError("refusing during partition")
            await self.replicate_quorum(k, v, peers)
            return True

        # AP: write locally, dispatch async hinted handoff
        self.local[k] = (v, self.vector_tick())
        asyncio.create_task(self.gossip(k))
        return True
// CP vs AP at the data tier — a strategy interface.
import { EventEmitter } from "node:events";

export interface Replication {
    write(k: string, v: string): Promise<boolean>;
}

export class CpReplication implements Replication {
    constructor(private readonly cluster: Cluster) {}

    async write(k: string, v: string): Promise<boolean> {
        if (!(await this.cluster.hasQuorum())) {
            throw new UnavailableError("partition — refusing");
        }
        return this.cluster.replicateMajority(k, v);
    }
}

export class ApReplication implements Replication {
    constructor(private readonly local: Map<string, Versioned>,
                private readonly vector: VectorClock,
                private readonly bus: EventEmitter) {}

    async write(k: string, v: string): Promise<boolean> {
        this.local.set(k, { value: v, clock: this.vector.tick() });
        setImmediate(() => this.bus.emit("gossip", k));   // async gossip
        return true;        // always succeed locally
    }
}
03

PACELC Theorem

Core Latency
Difficulty

The story

Daniel Abadi's extension to CAP, and arguably more useful day-to-day. CAP only tells you what happens during a partition — but partitions are rare. PACELC adds: else, when the system is healthy, you still trade off between Latency and Consistency. Want strong consistency? Pay round-trips for quorum. Want low latency? Read from a local replica that might be stale.

Cassandra is PA/EL — picks Availability under partition, Latency under normal ops. Spanner is PC/EC — picks Consistency in both modes (and pays for it with TrueTime atomic clocks). When someone asks "why is your DB slow on writes?" — this is the real answer.

Use when
  • Choosing between SQL and NoSQL stores
  • Justifying read replicas vs read-your-writes
  • Tuning consistency levels (ONE / QUORUM / ALL)
Avoid when
  • Talking to people who haven't met CAP yet
  • You can't articulate your latency budget
  • Bikeshedding instead of measuring
Connects to CAP, Consistency models, Quorums
// Cassandra-style tunable consistency — the EL/EC knob, per query.
type Consistency int
const (
    ONE    Consistency = iota   // fastest, weakest (EL)
    QUORUM                        // majority, balanced
    ALL                           // slowest, strongest (EC)
)

func (db *DB) Read(k string, c Consistency) (string, error) {
    need := map[Consistency]int{
        ONE: 1, QUORUM: db.N/2+1, ALL: db.N,
    }[c]
    replies := db.queryReplicas(k, need)

    // Pick latest version (last-write-wins by timestamp)
    winner := replies[0]
    for _, r := range replies[1:] {
        if r.Timestamp.After(winner.Timestamp) { winner = r }
    }
    go db.readRepair(k, winner, replies)   // async repair
    return winner.Value, nil
}
# Per-request consistency — the latency/consistency knob.
from enum import Enum

class CL(Enum):
    ONE    = 1     # lowest latency (~1 RTT) — EL
    QUORUM = 2     # N/2 + 1 — balanced
    ALL    = 3     # strongest, highest latency — EC

async def read(key: str, cl: CL = CL.ONE) -> str:
    N = replication_factor
    need = {CL.ONE: 1, CL.QUORUM: N // 2 + 1, CL.ALL: N}[cl]

    # Race the replicas. Return as soon as `need` agree (or disagree+timestamps).
    replies = await gather_first_n(need, [peer.read(key) for peer in peers])
    winner = max(replies, key=lambda r: r.timestamp)

    # Hint stale replicas to update — async, off the critical path.
    asyncio.create_task(read_repair(key, winner, replies))
    return winner.value
// cassandra-driver — consistency as a per-call concern.
import { Client, types } from "cassandra-driver";

const { consistencies } = types;

export class UserRepo {
    constructor(private readonly client: Client) {}

    // Hot path — speed first. Stale-tolerant (e.g. profile photo).
    async readFast(id: string): Promise<User> {
        const rs = await this.client.execute(
            "SELECT * FROM users WHERE id = ?", [id],
            { prepare: true, consistency: consistencies.one });
        return rs.first() as User;
    }

    // Critical — pay the latency cost (e.g. account balance).
    async readStrong(id: string): Promise<User> {
        const rs = await this.client.execute(
            "SELECT * FROM users WHERE id = ?", [id],
            { prepare: true, consistency: consistencies.quorum });
        return rs.first() as User;
    }

    // W=QUORUM + R=QUORUM = strong consistency. Two extra round-trips.
    async save(u: User): Promise<void> {
        await this.client.execute(
            "INSERT INTO users (id) VALUES (?)", [u.id],
            { prepare: true, consistency: consistencies.quorum });
    }
}
04

Consistency Models

Core Mental model
Difficulty

The story

There is no single "consistent" — there's a ladder, and you climb only as high as your invariants demand. Linearizable behaves as if every operation took effect at one instant on a single machine; you can compose objects safely. Sequential agrees on a global order but not real-time. Causal says "if A caused B, everyone sees A first" — perfect for chat and comments. Read-your-writes just promises you don't see your own changes disappear. Eventual promises convergence someday.

The senior move is picking the weakest level that preserves your invariants — every step up the ladder costs latency, throughput, or both. Comments on a post? Causal is plenty. Bank balance? Linearizable, no shortcuts.

Use when
  • Designing the consistency story for a feature
  • Debugging "I just wrote it but it's not there"
  • Justifying a choice of read replica strategy
Avoid when
  • You haven't written down your invariants
  • You assume strongest = best (it's slowest)
  • Marketing replaces measurement
Connects to CAP, PACELC, Isolation levels
// Read-your-writes via session stickiness to a write timestamp.
type Session struct {
    UserID      string
    LastWriteTS int64      // hybrid logical clock
}

func (db *DB) ReadRYW(s *Session, k string) (string, error) {
    // Try replicas — but require they've seen our last write.
    for _, r := range db.replicas {
        if r.appliedTS >= s.LastWriteTS {
            return r.read(k)         // safe: replica is caught up
        }
    }
    return db.leader.read(k)         // fall back to source of truth
}

func (db *DB) Write(s *Session, k, v string) error {
    ts := db.leader.put(k, v)
    s.LastWriteTS = ts             // caller now expects ≥ this ts
    return nil
}
# Causal consistency via dependency tokens.
# When you write, you get a token. Pass it to subsequent reads;
# they wait until the replica has applied everything you depend on.
from dataclasses import dataclass

@dataclass
class CausalToken:
    deps: dict[str, int]   # replica_id → last applied ts

async def write(k: str, v: str, tok: CausalToken) -> CausalToken:
    replica = pick_replica()
    await replica.wait_for(tok.deps)             # causal barrier
    ts = await replica.put(k, v)
    tok.deps[replica.id] = ts
    return tok

async def read(k: str, tok: CausalToken) -> str:
    replica = pick_replica()
    await replica.wait_for(tok.deps)             # cause-before-effect
    return await replica.get(k)
# MongoDB causal sessions, Cassandra LWT, and Cosmos all do variants of this.
// MongoDB Node driver — causal consistency via session tokens.
import { MongoClient } from "mongodb";

export class CommentService {
    constructor(private readonly mongo: MongoClient) {}

    // Same session = causal consistency: a follow-up read sees prior writes,
    // even when the read lands on a different secondary replica.
    async writeAndList(postId: string, c: Comment): Promise<Comment[]> {
        const s = this.mongo.startSession({ causalConsistency: true });
        try {
            const comments = this.mongo
                .db("app")
                .collection<Comment>("comments");

            await comments.insertOne(c, { session: s });

            return await comments
                .find({ postId }, { session: s, readPreference: "secondaryPreferred" })
                .toArray();
        } finally {
            await s.endSession();
        }
    }
    // Without the session: read might land on a stale secondary,
    // missing the very comment we just inserted.
}

Logical & Vector Clocks

Deep Theory
Difficulty

The story

Wall-clock time is unreliable across machines — clocks drift, NTP corrects in jumps, and "before" becomes a fiction. Lamport timestamps (1978) replace it with a counter: every event ticks the local clock, and every received message bumps your clock to max(local, sent) + 1. This gives you a total order — but loses information about concurrent events.

Vector clocks keep a counter per node. When you compare two vectors, you can see whether one happened-before the other or whether they're concurrent (neither dominates). This is the machinery underneath Dynamo, Riak, and most CRDT-based systems. The cost: vectors grow with the number of writers.

Use when
  • Detecting concurrent updates for conflict resolution
  • Building CRDTs or Dynamo-style stores
  • You can't trust wall-clock time across nodes
Avoid when
  • A single linearizable store would do
  • Writers are bounded and small (use Lamport)
  • Storage cost of growing vectors is prohibitive
Connects to Causal consistency, CQRS
// Vector clock — happens-before, concurrent, or strict descendant.
type VC map[string]int64

func (v VC) Tick(node string) { v[node]++ }

func (v VC) Merge(other VC) {
    for k, t := range other {
        if t > v[k] { v[k] = t }
    }
}

// Compare two vectors. Returns:
//  -1 = a happened-before b
//   1 = b happened-before a
//   0 = concurrent (irreconcilable without business logic)
func Compare(a, b VC) int {
    aLess, bLess := false, false
    for k := range union(a, b) {
        switch {
        case a[k] < b[k]: aLess = true
        case a[k] > b[k]: bLess = true
        }
    }
    switch {
    case aLess && !bLess: return -1
    case !aLess && bLess: return 1
    }
    return 0                 // concurrent
}
# Vector clock — minimal, complete.
from collections import defaultdict

class VectorClock:
    def __init__(self):
        self.v = defaultdict(int)

    def tick(self, node: str):
        self.v[node] += 1

    def merge(self, other: "VectorClock"):
        for k, t in other.v.items():
            self.v[k] = max(self.v[k], t)

    def compare(self, other: "VectorClock") -> str:
        keys = self.v.keys() | other.v.keys()
        a_less = any(self.v[k] < other.v[k] for k in keys)
        b_less = any(self.v[k] > other.v[k] for k in keys)
        if a_less and not b_less: return "before"
        if b_less and not a_less: return "after"
        if not a_less and not b_less: return "equal"
        return "concurrent"      # needs domain merge
// Vector clock — used in Riak/Dynamo-style conflict detection.

export type Order = "BEFORE" | "AFTER" | "EQUAL" | "CONCURRENT";

export class VectorClock {
    private readonly v = new Map<string, number>();

    tick(node: string): void {
        this.v.set(node, (this.v.get(node) ?? 0) + 1);
    }

    merge(other: VectorClock): void {
        for (const [k, t] of other.v) {
            this.v.set(k, Math.max(this.v.get(k) ?? 0, t));
        }
    }

    compare(o: VectorClock): Order {
        let aLess = false, bLess = false;
        const keys = new Set([...this.v.keys(), ...o.v.keys()]);
        for (const k of keys) {
            const a = this.v.get(k) ?? 0;
            const b = o.v.get(k) ?? 0;
            if (a < b) aLess = true;
            else if (a > b) bLess = true;
        }
        if (aLess && !bLess) return "BEFORE";
        if (!aLess && bLess) return "AFTER";
        if (!aLess && !bLess) return "EQUAL";
        return "CONCURRENT";        // needs domain merge
    }
}

Consensus & Raft

Core Algorithm
Difficulty

The story

If a single source of truth must be replicated across machines and survive failures, you need consensus. Paxos solved it first (1989) and broke an entire generation's brains; Raft (2014) solved it again with the explicit goal of being teachable. Both give you the same property: a cluster of N nodes can agree on a sequence of decisions as long as a majority is alive.

Raft's three pieces: leader election (followers timeout, become candidates, gather votes), log replication (the leader broadcasts entries; commits when a majority acks), and safety (a new leader can't be elected without the most up-to-date log). etcd, Consul, CockroachDB, TiKV, and most modern distributed databases run Raft underneath.

Use when
  • You need a strongly-consistent replicated state machine
  • You're building config storage, locks, or coordination
  • Loss of one node must not lose committed writes
Avoid when
  • You don't need strong consistency (use gossip)
  • Quorum latency is unacceptable for your hot path
  • You're tempted to write your own — etcd already exists
Connects to CAP, Quorums, Service Discovery
// Skeleton of Raft AppendEntries — the heart of log replication.
type Entry struct { Term uint64; Cmd []byte }

type AppendArgs struct {
    Term         uint64
    LeaderID     string
    PrevLogIndex uint64
    PrevLogTerm  uint64
    Entries      []Entry
    LeaderCommit uint64
}

func (r *Raft) AppendEntries(a AppendArgs) (uint64, bool) {
    // 1. Reply false if older term — protect against stale leaders
    if a.Term < r.currentTerm { return r.currentTerm, false }

    // 2. Step down if we see a higher term
    if a.Term > r.currentTerm {
        r.currentTerm = a.Term; r.role = Follower
    }
    r.resetElectionTimer()

    // 3. Log consistency check — refuse if our log diverges
    if !r.log.matches(a.PrevLogIndex, a.PrevLogTerm) {
        return r.currentTerm, false
    }

    // 4. Append new entries (truncating any conflict)
    r.log.appendFrom(a.PrevLogIndex+1, a.Entries)

    // 5. Advance commit index — apply to state machine
    if a.LeaderCommit > r.commitIndex {
        r.commitIndex = min(a.LeaderCommit, r.log.lastIndex())
        go r.applyCommitted()
    }
    return r.currentTerm, true
}
# Raft AppendEntries handler. Use a real impl in production:
# pyraft, python-raft, or wrap etcd's gRPC.
from dataclasses import dataclass

@dataclass
class AppendArgs:
    term:           int
    leader_id:      str
    prev_log_idx:   int
    prev_log_term:  int
    entries:        list
    leader_commit:  int

class Raft:
    def append_entries(self, a: AppendArgs) -> tuple[int, bool]:
        if a.term < self.current_term:
            return self.current_term, False           # reject stale leader

        if a.term > self.current_term:
            self.current_term, self.role = a.term, "follower"

        self.reset_election_timer()

        if not self.log.matches(a.prev_log_idx, a.prev_log_term):
            return self.current_term, False           # gap or fork

        self.log.append_from(a.prev_log_idx + 1, a.entries)

        if a.leader_commit > self.commit_index:
            self.commit_index = min(a.leader_commit, self.log.last_idx())
            self.apply_committed()                  # to state machine

        return self.current_term, True

07

Quorums · R + W > N

Core Consistency
Difficulty

The story

If you replicate a value across N nodes, write to W, and read from R, the quorum trick says: as long as R + W > N, every read intersects every write by at least one node. That's how Dynamo, Cassandra, and Riak get consistency on demand without a leader.

The math gives you a tunable knob. N=5, W=3, R=3 means strong consistency with two-node failure tolerance. N=5, W=1, R=1 is fast but can return stale reads. W=N is durable but blocks on any one node being slow. Senior engineers tune this per workload — analytics tolerates stale reads, accounting doesn't.

Use when
  • Tuning a Dynamo-style or Cassandra deployment
  • Trading consistency for availability per query
  • Designing leaderless replication
Avoid when
  • Writes are strictly ordered (use Raft)
  • Network is heavily partitioned often
  • You need cross-key transactions (quorums are per-key)
Connects to CAP, PACELC, Raft
// Quorum read — wait for R replicas to reply, return latest version.
type Versioned struct { V string; TS int64 }

func (c *Cluster) Get(k string, R int) (Versioned, error) {
    replies := make(chan Versioned, len(c.replicas))
    for _, r := range c.replicas {
        go func(r *Replica) {
            if v, err := r.Get(k); err == nil { replies <- v }
        }(r)
    }
    var best Versioned
    for i := 0; i < R; i++ {
        select {
        case v := <-replies:
            if v.TS > best.TS { best = v }
        case <-time.After(c.timeout):
            return best, errors.New("quorum read timeout")
        }
    }
    return best, nil
}
# Quorum write — accept after W replicas have ack'd. Race the rest.
import asyncio, time

async def quorum_put(cluster, k: str, v: str, W: int) -> bool:
    ts = time.time_ns()
    payload = {"k": k, "v": v, "ts": ts}
    tasks = [asyncio.create_task(r.put(payload)) for r in cluster.replicas]

    acks = 0
    for coro in asyncio.as_completed(tasks, timeout=2.0):
        try:
            await coro
            acks += 1
            if acks >= W:
                return True      # quorum reached — others continue
        except Exception:
            continue              # count successes only
    return False              # quorum not reached

# R + W > N: pick the right knob for the right call.
# Idempotent retries make >W writes harmless.

        await this.client.execute(
            "UPDATE accounts SET balance = balance + ? WHERE id = ?",
            [delta, id],
            { prepare: true, consistency: types.consistencies.quorum });
    }

    // R = QUORUM. Combined with W=QUORUM you get strong reads.
    // Combined with W=ONE you get fast reads but possibly stale.
    async balance(id: string): Promise<number> {
        const rs = await this.client.execute(
            "SELECT balance FROM accounts WHERE id = ?",
            [id],
            { prepare: true, consistency: types.consistencies.quorum });
        return Number(rs.first().get("balance"));
    }
}

Idempotency · the practical answer

Core Pattern
Difficulty

The story

Most of distributed systems comes down to one survival skill: make every operation idempotent. If a request can be retried any number of times and produce the same effect as one execution, then partial failures, timeouts, network glitches, and duplicated messages stop being correctness bugs — they become latency.

Two flavors. Naturally idempotent operations (PUT a state, set a value) are free. Effectfully idempotent operations (charge a card, send an email) require an Idempotency-Key the client picks before the first attempt — the server stores the key + result, and replays the stored reply on retry. This is the pattern Stripe, AWS, and every payment system relies on.

Use when
  • Any operation with side effects that could be retried
  • Webhook receivers, payment endpoints, message consumers
  • You want safe at-least-once instead of mythical exactly-once
Avoid when
  • Operations that must run more than once (counters)
  • Storage cost of keeping keys is prohibitive (use TTL)
  • Keys leak across customers — namespace them
Connects to Idempotency pattern, Exactly-once, Outbox
// Idempotency middleware — store key → response, replay on duplicate.
func Idempotent(store KVStore, h http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        key := r.Header.Get("Idempotency-Key")
        if key == "" { h.ServeHTTP(w, r); return }

        // Reserve atomically. If already taken — replay.
        _, created, _ := store.SetIfAbsent(key, "in-flight", 24*time.Hour)
        if !created {
            if resp, ok := store.Get(key); ok && resp != "in-flight" {
                writeStored(w, resp); return
            }
            http.Error(w, "in flight", 409); return
        }

        // First time — run the handler, capture, store.
        rec := &recorder{ResponseWriter: w}
        h.ServeHTTP(rec, r)
        store.Set(key, rec.snapshot(), 24*time.Hour)
    })
}

from fastapi import FastAPI, Header, HTTPException, Depends
import json, redis.asyncio as aioredis

app = FastAPI()
r = aioredis.from_url("redis://localhost")

async def idempotent(key: str | None = Header(None, alias="Idempotency-Key")):
    if not key: return None

    cache_key = f"idem:{key}"
    cached = await r.get(cache_key)
    if cached == b"in-flight":
        raise HTTPException(409, "request in flight")
    if cached:
        return {"replay": json.loads(cached)}

    # Reserve. If we lose the race, fall through to replay path next call.
    if not await r.set(cache_key, b"in-flight", ex=86400, nx=True):
        raise HTTPException(409, "in flight")
    return cache_key

import type { Request, Response, NextFunction } from "express";
import { Redis } from "ioredis";

const redis = new Redis();
const DAY = 24 * 60 * 60;

export async function idempotency(req: Request, res: Response, next: NextFunction) {
    const key = req.get("Idempotency-Key");
    if (!key) return next();

    const k = `idem:${key}`;
    const cached = await redis.get(k);
    if (cached === "in-flight") {
        return res.status(409).send("request in flight");
    }
    if (cached !== null) {
        return res.status(200).type("json").send(cached);
    }
    if ((await redis.set(k, "in-flight", "EX", DAY, "NX")) === null) {
        return res.status(409).send("in flight");
    }
    // Wrap send() so the handler's response body is cached for replays.
    const send = res.send.bind(res);
    res.send = (body?: unknown) => {
        void redis.set(k, String(body), "EX", DAY);
        return send(body);
    };
    next();
}

Exactly-Once is a lie

Myth Theory
Difficulty

The story

"Exactly-once delivery" is marketing. The Two Generals Problem proves it can't exist over a lossy channel: if the sender doesn't know the receiver got it, retrying might duplicate; not retrying might lose. There is no protocol that delivers a single message exactly once across a real network.

What you can have is exactly-once processing: build at-least-once delivery (retry until ack) and pair it with idempotent consumers (deduplicate by key, or atomically commit message-offset + side-effect together). Kafka transactions, Flink checkpointing, and the Outbox pattern all amount to this. Anyone selling you "exactly-once" is selling you this trick — make sure they say so.

Use when
  • You need exactly-once effects (not delivery)
  • Stream processing with strict accounting semantics
  • Event-driven payment / billing systems
Avoid when
  • Someone tells you a queue gives you exactly-once — push back
  • You haven't paired delivery with idempotency
  • Best-effort fire-and-forget is genuinely fine
Connects to Idempotency, Outbox, Event-driven
// At-least-once delivery + idempotent consumer = exactly-once processing.
func consume(msg kafka.Message, db *sql.DB) error {
    tx, _ := db.Begin()
    defer tx.Rollback()

    // 1. Dedup table — UNIQUE(message_id) is the safety net
    _, err := tx.Exec(
        `INSERT INTO processed_messages(id, processed_at)
         VALUES($1, NOW()) ON CONFLICT DO NOTHING`,
        msg.Key,
    )
    if err != nil { return err }

    // 2. Was this a duplicate?
    var rows int64
    tx.QueryRow("SELECT changes()").Scan(&rows)
    if rows == 0 {
        return tx.Commit()    // already processed — ack and skip
    }

    // 3. Apply the side effect in the SAME transaction
    if err := applyBusinessLogic(tx, msg); err != nil { return err }

    // 4. Commit. Ack to Kafka only AFTER commit succeeds.
    return tx.Commit()
}

            throw err;                                   // no ack → redelivered
        } finally {
            client.release();
        }
        // 3. Transaction committed. Ack (commit the offset) only after.
        const offset = (BigInt(message.offset) + 1n).toString();
        await consumer.commitOffsets([{ topic, partition, offset }]);
    },

Failure Detectors & Heartbeats

Core Practical
Difficulty

The story

You can't tell a dead node from a slow one without a timeout — and any timeout is a guess. Set it too low: false alarms during GC pauses or network blips. Set it too high: real failures linger, calls hang, queues back up. Failure detectors abstract this trade-off into something tunable.

The naive approach is fixed timeouts and missed-heartbeat counts. The better approach is the phi accrual failure detector (Hayashibara et al, 2004): track inter-arrival times and emit a continuous suspicion value φ. Cassandra and Akka use it. The key insight: failures are statistical, so detection should be too.

Use when
  • Cluster membership and gossip protocols
  • Service discovery health checks
  • Leader election timeouts (Raft)
Avoid when
  • You can rely on a managed health system (k8s)
  • Latency variance is so high signal is meaningless
  • Two services where a fixed timeout is plenty
Connects to Raft, Circuit Breaker, Service Discovery
// Phi accrual detector — tracks heartbeat arrival distribution.
type PhiDetector struct {
    arrivals  []time.Time      // recent arrival timestamps
    window    int              // keep the last N samples
    threshold float64          // e.g. φ ≥ 8 → suspect
}

func (d *PhiDetector) Heartbeat() {
    d.arrivals = append(d.arrivals, time.Now())
    if len(d.arrivals) > d.window {
        d.arrivals = d.arrivals[1:]
    }
}

// Compute "suspicion level" — higher = more likely down
func (d *PhiDetector) Phi(now time.Time) float64 {
    if len(d.arrivals) < 2 { return 0 }
    last := d.arrivals[len(d.arrivals)-1]
    elapsed := now.Sub(last).Seconds()

    mean, stddev := stats(d.arrivals)
    p := 1 - cdf(elapsed, mean, stddev)   // tail probability
    if p < 1e-12 { p = 1e-12 }
    return -math.Log10(p)
}

func (d *PhiDetector) IsSuspect() bool {
    return d.Phi(time.Now()) >= d.threshold
}
# Simple moving-window heartbeat detector — close cousin of phi accrual.
import time, math, statistics
from collections import deque

class FailureDetector:
    def __init__(self, window=200, threshold=8.0):
        self.arrivals = deque(maxlen=window)
        self.threshold = threshold

    def heartbeat(self):
        self.arrivals.append(time.monotonic())

    def phi(self) -> float:
        if len(self.arrivals) < 2:
            return 0.0
        elapsed = time.monotonic() - self.arrivals[-1]
        intervals = [b - a for a, b in zip(self.arrivals, list(self.arrivals)[1:])]
        μ = statistics.mean(intervals)
        σ = max(statistics.stdev(intervals), 1e-3)

        # P(message arrives later than `elapsed` given the distribution)
        z = (elapsed - μ) / σ
        p = max(0.5 * math.erfc(z / math.sqrt(2)), 1e-12)
        return -math.log10(p)

    def is_suspect(self) -> bool:
        return self.phi() >= self.threshold
// Akka's phi accrual detector — production-grade, used in Cassandra too.
import { erf } from "mathjs";

export class PhiAccrualDetector {
    private readonly arrivals: number[] = [];

    constructor(private readonly window: number,
                private readonly threshold: number) {}   // 8.0 is a common default

    // One event loop, no threads: no `synchronized` needed.
    heartbeat(): void {
        this.arrivals.push(performance.now());
        while (this.arrivals.length > this.window) this.arrivals.shift();
    }

    phi(): number {
        if (this.arrivals.length < 2) return 0;
        const last = this.arrivals[this.arrivals.length - 1];
        const elapsedMs = performance.now() - last;

        const [mean, sd] = computeStats(this.arrivals);
        const stddev = Math.max(sd, 0.001);
        const z = (elapsedMs - mean) / stddev;
        const p = Math.max(0.5 * (1 - erf(z / Math.SQRT2)), 1e-12);
        return -Math.log10(p);
    }

    isSuspect(): boolean { return this.phi() >= this.threshold; }
}
