01

Indexes · B-tree vs LSM

Core Storage
Difficulty

The story

B-trees have ruled databases since the 1970s: a balanced tree of disk pages, lookups in O(log N), updates in place. They optimise for reads — the page you want is exactly where it should be. The cost is random I/O on writes: you go find a page, modify it, fsync it back. Postgres, MySQL/InnoDB, Oracle, SQL Server all use them.

LSM-trees (log-structured merge) flip the trade. New writes append to an in-memory memtable, which periodically flushes to a sorted file (SSTable) on disk. Reads might check several levels, but writes are pure sequential I/O — orders of magnitude faster on spinning disks and SSDs alike. RocksDB, Cassandra, ScyllaDB, ClickHouse, and most "fast write" systems use LSM. The price is read amplification and background compaction.

B-tree when
  • Read-heavy with low write rate
  • Range scans dominate workload
  • Predictable, low write amplification matters
LSM when
  • Write-heavy logs, time series, IoT
  • Sequential writes >> random reads
  • You can spend SSD on read amplification
Connects to Query Plans, MVCC, Caching
// Picking the right index for the query, not the other way around.
var migrations = []string{
    // B-tree: default, used for equality + range scans
    `CREATE INDEX idx_orders_user_created
       ON orders(user_id, created_at DESC);`,

    // Partial index: only index "open" rows — smaller, faster, hotter cache
    `CREATE INDEX idx_orders_open
       ON orders(user_id) WHERE status = 'open';`,

    // Covering index: query reads everything from the index, never the table
    `CREATE INDEX idx_orders_covering
       ON orders(user_id, created_at DESC) INCLUDE (total, status);`,

    // GIN: full-text or array containment — wrong tool for ranges
    `CREATE INDEX idx_orders_tags ON orders USING GIN (tags);`,

    // BRIN: tiny index for huge append-only tables (time series)
    `CREATE INDEX idx_events_time ON events USING BRIN (occurred_at);`,
}

// Rules of thumb:
// — One index per query pattern, not per column
// — Leftmost prefix wins on composite indexes
// — Every extra index taxes writes; measure before adding
# SQLAlchemy — declaring the right index for each query pattern.
from sqlalchemy import Column, Integer, DateTime, Index, String
from sqlalchemy.dialects.postgresql import JSONB

class Order(Base):
    __tablename__ = "orders"

    id         = Column(Integer, primary_key=True)
    user_id    = Column(Integer, nullable=False)
    status     = Column(String, nullable=False)
    total      = Column(Integer, nullable=False)
    tags       = Column(JSONB)
    created_at = Column(DateTime, nullable=False)

    __table_args__ = (
        # B-tree composite (leftmost prefix rule)
        Index("idx_orders_user_created",
              "user_id", created_at.desc()),

        # Partial: only "open" rows — saves a lot of cache
        Index("idx_orders_open", "user_id",
              postgresql_where=(status == "open")),

        # GIN over JSONB tags — equality / containment, not ranges
        Index("idx_orders_tags", "tags", postgresql_using="gin"),
    )

# Each extra index slows writes. Drop the ones EXPLAIN doesn't use.
// Flyway / JPA — declaring indexes that match the query patterns.

@Entity

        @Index(name = "idx_orders_status", columnList = "status")

public class Order {

    @Column(nullable = false) private String status;
    @Column(nullable = false) private Long total;

02

Query Plans · the EXPLAIN reflex

Core Practical
Difficulty

The story

Most queries that "the database is slow" turn out to be queries the optimiser ran badly — full sequential scans where you expected an index, hash joins where a merge would have been cheaper, or a wildly wrong row estimate that picked the wrong plan. Senior engineers don't guess; they run EXPLAIN ANALYZE.

The skill is reading the plan tree from the bottom up. Each node tells you how many rows it read, how long it took, and what kind of access it used. Seq Scan on a big table is almost always the fix-target. Wrong row estimates often mean stale stats — an ANALYZE away from a 100× speed-up. EXPLAIN is the highest-leverage tool you'll ever learn for backend work.

Use when
  • Any query > 100ms in production logs
  • Before adding an index — what's the plan say?
  • After a schema change — did the optimiser still pick well?
Avoid when
  • Trusting EXPLAIN without ANALYZE — estimates lie
  • Optimising a query nobody actually runs
  • Over-indexing because EXPLAIN shows a Seq Scan once
Connects to Indexes, N+1 problem
// Programmatic EXPLAIN — useful in slow-query logging middleware.
func explainSlow(db *sql.DB, query string, args ...interface{}) string {
    rows, err := db.Query(
        "EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON) "+query, args...)
    if err != nil { return "" }
    defer rows.Close()

    var plan string
    for rows.Next() {
        var line string
        rows.Scan(&line)
        plan += line + "\n"
    }
    return plan
}

// Wrap your DB calls; log the plan whenever a query exceeds threshold.
func (d *DB) QuerySlow(q string, args ...interface{}) (*sql.Rows, error) {
    start := time.Now()
    rows, err := d.db.Query(q, args...)
    if elapsed := time.Since(start); elapsed > 200*time.Millisecond {
        log.Warn("slow_query",
            "sql", q, "ms", elapsed.Milliseconds(),
            "plan", explainSlow(d.db, q, args...))
    }
    return rows, err
}
# Read the plan, log it on slow queries.
import json, time, logging
import asyncpg

log = logging.getLogger(__name__)

async def explain(conn: asyncpg.Connection, sql: str, *args) -> dict:
    rows = await conn.fetch(
        f"EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON) {sql}", *args)
    return json.loads(rows[0][0])[0]["Plan"]

async def query_with_explain(conn, sql: str, *args, slow_ms=200):
    t0 = time.monotonic()
    rows = await conn.fetch(sql, *args)
    elapsed_ms = (time.monotonic() - t0) * 1000
    if elapsed_ms >= slow_ms:
        plan = await explain(conn, sql, *args)
        log.warning(
            "slow_query",
            extra={"sql": sql, "ms": elapsed_ms,
                   "node": plan["Node Type"],
                   "actual_rows": plan["Actual Rows"],
                   "estimated_rows": plan["Plan Rows"]},
        )
    return rows

# Estimated vs actual mismatch > 10× → ANALYZE the table or rewrite the query.
// Hibernate / Spring — log the plan when a query goes slow.

@Aspect
@Component
public class SlowQueryAdvice {

    private final JdbcTemplate jdbc;
    private final long slowMs = 200;

    @Around("@annotation(org.springframework.transaction.annotation.Transactional)")
    public Object timeIt(ProceedingJoinPoint pjp) throws Throwable {
        long t0 = System.nanoTime();
        Object out = pjp.proceed();
        long ms = (System.nanoTime() - t0) / 1_000_000;
        if (ms >= slowMs) {
            log.warn("slow_op {} ms — run EXPLAIN ANALYZE on the offending query", ms);
        }
        return out;
    }

    // Helper to fetch the plan when investigating
    public String explain(String sql, Object... args) {
        List<String> lines = jdbc.query(
            "EXPLAIN (ANALYZE, BUFFERS) " + sql,
            (rs, i) -> rs.getString(1), args);
        return String.join("\n", lines);
    }
}
03

ACID & the I in it

Core Theory
Difficulty

The story

The ACID contract is the foundation of relational databases — but most engineers learn the acronym and stop there. Atomicity means a transaction is all or nothing (the WAL gives you this). Consistency means invariants hold across the transaction (FKs, check constraints). Durability means a committed write survives a crash (fsync after WAL). These three are mostly handled for you.

Isolation is where the bugs are. It's not a yes/no property — it's a spectrum (next concept). Most databases default to a weaker isolation than people assume. The "race condition" you blame on async code is often a Read Committed anomaly that wouldn't exist at Serializable. Senior engineers know which level they're running at and which anomalies that lets through.

Use when
  • Anything money-related needs ACID — full stop
  • Your workflow has multi-row invariants
  • You need durable, atomic state changes
Avoid when
  • Pretending BASE / NoSQL is ACID — it isn't
  • Thinking "the database handles it" without checking I
  • Assuming durability without verifying fsync settings
Connects to Isolation levels, MVCC, Consistency models
// A real ACID transaction — atomic, durable, with explicit isolation.
func transferFunds(db *sql.DB, from, to int64, cents int64) error {
    // Pick the level explicitly — defaults vary across drivers
    tx, err := db.BeginTx(ctx, &sql.TxOptions{
        Isolation: sql.LevelSerializable,
    })
    if err != nil { return err }
    defer tx.Rollback()                // no-op if commit succeeded

    // A: both updates land or neither do
    if _, err = tx.Exec(
        "UPDATE accounts SET balance = balance - $1 WHERE id = $2",
        cents, from); err != nil {
        return err
    }
    if _, err = tx.Exec(
        "UPDATE accounts SET balance = balance + $1 WHERE id = $2",
        cents, to); err != nil {
        return err
    }

    // C: CHECK (balance >= 0) constraint catches overdrafts
    // D: commit forces WAL fsync — survives a crash
    return tx.Commit()
}
# SQLAlchemy with explicit isolation level.
from sqlalchemy import create_engine, text
from sqlalchemy.exc import OperationalError

# Connection-level: every transaction here runs SERIALIZABLE.
engine = create_engine("postgresql+psycopg://...",
                       isolation_level="SERIALIZABLE")

def transfer(from_id: int, to_id: int, cents: int):
    # Retry serialization failures (40001) — it's expected at this level
    for attempt in range(3):
        try:
            with engine.begin() as conn:
                conn.execute(text("""
                    UPDATE accounts SET balance = balance - :c
                    WHERE id = :from
                """), {"c": cents, "from": from_id})
                conn.execute(text("""
                    UPDATE accounts SET balance = balance + :c
                    WHERE id = :to
                """), {"c": cents, "to": to_id})
                # commit on context exit; rollback on any exception
            return
        except OperationalError as e:
            if "40001" in str(e.orig) and attempt < 2:
                continue           # serialization failure — retry
            raise

# C is enforced by CHECK (balance >= 0). D is the engine's WAL fsync.
// Spring's @Transactional makes ACID the default.

@Service
public class TransferService {

    private final JdbcTemplate jdbc;

    @Transactional(isolation = Isolation.SERIALIZABLE,
                   rollbackFor = Exception.class)
    @Retryable(
        retryFor = CannotSerializeTransactionException.class,
        maxAttempts = 3,
        backoff = @Backoff(delay = 25, multiplier = 2)
    )
    public void transfer(long from, long to, long cents) {
        // A: both run or neither — Spring handles rollback on exception
        jdbc.update("UPDATE accounts SET balance = balance - ? WHERE id = ?",
                     cents, from);
        jdbc.update("UPDATE accounts SET balance = balance + ? WHERE id = ?",
                     cents, to);
        // C: CHECK constraint catches negatives. Will throw → rollback.
        // D: commit forces WAL fsync. Crash-safe.
        // I: SERIALIZABLE — concurrent transfers serialized cleanly.
    }
}
04

Isolation Levels & Anomalies

Advanced Core
Difficulty

The story

SQL defines four isolation levels — and a list of "anomalies" each level allows. Dirty read: you see another transaction's uncommitted write. Non-repeatable read: read the same row twice in a transaction, get different values. Phantom: re-run a range query and new rows have appeared. Write skew: two transactions read the same data, make decisions based on it, and commit changes that together violate an invariant — even though each was correct in isolation.

The trap: defaults are weak. Postgres defaults to Read Committed; MySQL/InnoDB defaults to Repeatable Read. Most "this can't happen" bugs are the higher anomalies sneaking through. The fix is usually SELECT FOR UPDATE, or bumping to SERIALIZABLE for the path that matters and retrying on serialization failure.

Use when
  • Money, inventory, scheduling — anywhere invariants matter
  • Diagnosing "race condition" bugs in DB code
  • Designing concurrent-update flows
Avoid when
  • Cranking everything to Serializable globally — too slow
  • Trusting your ORM's default — check the actual session
  • Believing "Repeatable Read" means the same in every DB
Connects to MVCC, ACID, Consistency models
// The classic write-skew bug: two doctors going off-call at once.
// Read Committed lets it through; Serializable catches it.
func goOffCall(db *sql.DB, doctorID int64) error {
    tx, _ := db.BeginTx(ctx, &sql.TxOptions{
        Isolation: sql.LevelSerializable,
    })
    defer tx.Rollback()

    // Read: how many doctors are currently on-call?
    var onCall int
    tx.QueryRow(
        "SELECT count(*) FROM doctors WHERE on_call = true",
    ).Scan(&onCall)

    // Decision based on the read
    if onCall <= 1 {
        return errors.New("can't go off — you're the last")
    }
    // Write based on the decision
    _, err := tx.Exec(
        "UPDATE doctors SET on_call = false WHERE id = $1", doctorID)
    if err != nil { return err }

    // At Read Committed, two doctors race here and both go off-call.
    // At Serializable, the second commit fails with 40001 — retry it.
    return tx.Commit()
}
# Pessimistic locking: SELECT ... FOR UPDATE locks the rows you read.
from sqlalchemy import text

def withdraw(conn, account_id: int, cents: int) -> bool:
    # FOR UPDATE = "lock this row until I commit/rollback".
    # Other transactions trying to lock the same row block here.
    row = conn.execute(text("""
        SELECT balance FROM accounts
         WHERE id = :id
         FOR UPDATE
    """), {"id": account_id}).fetchone()

    if row.balance < cents:
        return False                          # insufficient funds

    conn.execute(text("""
        UPDATE accounts SET balance = balance - :c
         WHERE id = :id
    """), {"c": cents, "id": account_id})
    return True

# Optimistic alternative — version column, no locks held during the work:
#   UPDATE accounts SET balance = balance - :c, version = version + 1
#    WHERE id = :id AND version = :v   -- if 0 rows changed → retry
// Spring + JPA: optimistic locking via @Version, retry on conflict.

@Entity
public class Account {
    @Id      private Long id;
    @Version private Long version;          // JPA bumps it; mismatch → ex.
    private          Long balance;
    // getters/setters elided
}

@Service
public class AccountService {

    private final AccountRepo repo;

    @Transactional
    @Retryable(
        retryFor = ObjectOptimisticLockingFailureException.class,
        maxAttempts = 5,
        backoff = @Backoff(delay = 10, multiplier = 2)
    )
    public boolean withdraw(Long id, long cents) {
        Account a = repo.findById(id).orElseThrow();
        if (a.getBalance() < cents) return false;
        a.setBalance(a.getBalance() - cents);
        repo.save(a);                  // commit checks @Version
        return true;
    }
    // Pessimistic alternative: @Lock(LockModeType.PESSIMISTIC_WRITE) on the query.
}
05

MVCC · readers don't block writers

Advanced Engine
Difficulty

The story

MVCC (Multi-Version Concurrency Control) is the engine that makes "readers don't block writers, writers don't block readers" possible in Postgres, MySQL/InnoDB, Oracle, and SQL Server's snapshot isolation. Each row has visibility metadata — xmin (creating txn) and xmax (deleting txn). When you start a transaction, you get a snapshot ID; you only see versions whose visibility range covers your snapshot.

The cost is bloat: dead row versions pile up until vacuum/purge cleans them. Long-running transactions block vacuum, so a 4-hour analytics query can starve OLTP. The other cost is visibility checks at read time. The benefit is enormous concurrency — and a clean way to implement Repeatable Read and Snapshot Isolation without read locks.

Use when
  • Default for any modern OLTP database
  • Heavy concurrent reads + occasional writes
  • You need point-in-time consistency for reports
Watch out for
  • Long-running transactions causing bloat
  • Vacuum/purge tuning being neglected
  • Update-heavy hot rows fragmenting the heap
Connects to Isolation levels, Logical clocks
// Find long-running transactions that block vacuum (the MVCC enemy).
func findVacuumBlockers(db *sql.DB) ([]Blocker, error) {
    rows, err := db.Query(`
        SELECT pid,
               now() - xact_start    AS age,
               state,
               left(query, 80)       AS sample
          FROM pg_stat_activity
         WHERE xact_start IS NOT NULL
           AND now() - xact_start > interval '5 minutes'
         ORDER BY age DESC
    `)
    if err != nil { return nil, err }
    defer rows.Close()

    var out []Blocker
    for rows.Next() {
        var b Blocker
        rows.Scan(&b.PID, &b.Age, &b.State, &b.Sample)
        out = append(out, b)
    }
    return out, nil
}

06

Replication topologies

Core Architecture
Difficulty

The story

Leader-follower is the default in SQL world: one node accepts writes, asynchronously streams them to read replicas. Simple, strongly consistent at the leader, eventually consistent at the followers. Failover is the hard part — promote a follower, redirect writes, hope you didn't lose un-replicated commits.

Multi-leader lets you write at any region (good for write latency, multi-DC active-active) but you now have conflicts when two leaders accept conflicting writes. You need conflict resolution: last-write-wins, CRDTs, or manual reconciliation. Leaderless (Cassandra, Riak) ditches the leader entirely — clients write to N replicas and require quorum overlap (R + W > N) for consistency.

Pick when
  • L→F: standard SQL, scaling reads
  • Multi-leader: multi-region writes, geographic resilience
  • Leaderless: massive write throughput, can tolerate eventual
Avoid when
  • Reading from followers and expecting RYW (use sync replicas)
  • Multi-leader without a real conflict resolution plan
  • Leaderless when you need cross-key transactions
Connects to Quorums, CAP, Sharding
// Read/write split — writes to primary, reads to followers.
type DB struct {
    primary   *sql.DB
    replicas  []*sql.DB
    next      atomic.Uint64     // round-robin counter
}

func (d *DB) Read(ctx context.Context, q string, a ...interface{}) (*sql.Rows, error) {
    i := d.next.Add(1) % uint64(len(d.replicas))
    return d.replicas[i].QueryContext(ctx, q, a...)
}

// Read-your-writes: stick to primary briefly after a write
func (d *DB) ReadAfterWrite(ctx context.Context, key string) bool {
    // If the user wrote in the last 2s, route reads to primary
    return time.Since(d.lastWrite(key)) < 2*time.Second
}

func (d *DB) Get(ctx context.Context, user string, q string) (*sql.Rows, error) {
    if d.ReadAfterWrite(ctx, user) {
        return d.primary.QueryContext(ctx, q)
    }
    return d.Read(ctx, q)
}
# Read/write split with replication-lag awareness.
import random, time
import asyncpg

class SplitDB:
    def __init__(self, primary, replicas):
        self.primary = primary
        self.replicas = replicas
        self.last_write: dict[str, float] = {}

    async def write(self, user_id: str, sql: str, *args):
        res = await self.primary.execute(sql, *args)
        self.last_write[user_id] = time.monotonic()
        return res

    async def read(self, user_id: str, sql: str, *args):
        # Read-your-writes window — go to primary if recent
        recent = time.monotonic() - self.last_write.get(user_id, 0)
        if recent < 2.0:
            return await self.primary.fetch(sql, *args)
        # Otherwise: random replica with circuit-breaker fallback
        replica = random.choice(self.replicas)
        try:
            return await replica.fetch(sql, *args)
        except Exception:
            return await self.primary.fetch(sql, *args)
// Spring + AbstractRoutingDataSource — write to primary, read from replica.

07

Sharding strategies

Core Scale
Difficulty

The story

One database can scale very far on modern hardware — but eventually you hit a ceiling on writes, table size, or index memory. Sharding partitions the data across N machines. The pivot question is the shard key: pick wrong and you get hot shards forever.

Range sharding is great for ordered queries but creates hot shards (everyone writes to the latest range). Hash sharding spreads load evenly but kills range scans. Directory sharding (a lookup table) is flexible — perfect for multi-tenant — but adds a hop and another thing to keep consistent. Consistent hashing is the trick that makes resharding tolerable: adding a node moves only 1/N of the keys, not all of them.

Use when
  • You've outgrown a single primary write node
  • Data has natural partition boundaries (tenant, region)
  • Working set exceeds RAM on the biggest box you can buy
Avoid when
  • You haven't tuned indexes / vacuum on a single node yet
  • Cross-shard transactions dominate your workload
  • You don't have a story for resharding (you will need one)
Connects to Replication, Quorums, Gateway
// Consistent hashing — adding/removing a node moves ~1/N of keys.
type Ring struct {
    nodes    map[uint32]string     // hash → node id
    sorted   []uint32              // for binary search
    replicas int                   // virtual nodes per real node
}

func (r *Ring) Add(node string) {
    for i := 0; i < r.replicas; i++ {
        h := hash(fmt.Sprintf("%s#%d", node, i))
        r.nodes[h] = node
        r.sorted = append(r.sorted, h)
    }
    sort.Slice(r.sorted, func(i, j int) bool {
        return r.sorted[i] < r.sorted[j]
    })
}

func (r *Ring) Pick(key string) string {
    if len(r.sorted) == 0 { return "" }
    h := hash(key)
    // First node clockwise on the ring
    i := sort.Search(len(r.sorted), func(i int) bool {
        return r.sorted[i] >= h
    })
    if i == len(r.sorted) { i = 0 }   // wrap around
    return r.nodes[r.sorted[i]]
}
# Application-level sharding — pick a shard, route the query.
import hashlib
from bisect import bisect

class Ring:
    def __init__(self, vnodes=128):
        self.vnodes = vnodes
        self.ring: dict[int, str] = {}
        self.sorted: list[int] = []

    def _hash(self, s: str) -> int:
        return int(hashlib.md5(s.encode()).hexdigest(), 16)

    def add(self, node: str):
        for i in range(self.vnodes):
            h = self._hash(f"{node}#{i}")
            self.ring[h] = node
        self.sorted = sorted(self.ring.keys())

    def pick(self, key: str) -> str:
        if not self.sorted:
            raise RuntimeError("empty ring")
        h = self._hash(key)
        i = bisect(self.sorted, h) % len(self.sorted)
        return self.ring[self.sorted[i]]

08

Change Data Capture (CDC)

Core Pattern
Difficulty

The story

Every relational database already keeps a log of every change it makes — the WAL (Postgres), binlog (MySQL), or redo log. CDC reads that log and emits a stream of events to anyone who wants to know what changed. Debezium is the canonical implementation; it publishes to Kafka in a structured format with before/after images, transaction boundaries, and schema metadata.

Why care: CDC eliminates the dual-write problem. Instead of writing to your DB and then publishing an event (which can fail mid-way and leave the system inconsistent), you write to the DB and CDC publishes the event for you, atomically tied to the commit. Search indexes, analytics warehouses, cache invalidation, and event-driven workflows all become much simpler. The downside: ordering across tables, schema evolution, and long-running CDC consumers need real ops attention.

Use when
  • Streaming DB changes to search / analytics / caches
  • Eliminating dual-write inconsistency
  • Decoupling read models from the source of truth
Avoid when
  • You need transactional event semantics — use Outbox
  • Your team can't operate Kafka + Connect at scale
  • The downstream needs decisions, not just changes
Connects to Outbox, Event-driven, Dual writes
// Postgres logical replication consumer — reads change events directly.
import "github.com/jackc/pglogrepl"

func consumeWAL(ctx context.Context, conn *pgconn.PgConn) error {
    // Start replication from the slot we created earlier
    err := pglogrepl.StartReplication(ctx, conn,
        "orders_slot", 0, pglogrepl.StartReplicationOptions{
            PluginArgs: []string{
                "proto_version '2'",
                "publication_names 'orders_pub'",
            },
        })
    if err != nil { return err }

    for {
        msg, err := conn.ReceiveMessage(ctx)
        if err != nil { return err }

        // Decode the WAL message — INSERT/UPDATE/DELETE w/ before+after
        change, _ := decode(msg)
        publishToKafka("orders.changes", change)

        // CRITICAL: only ack after Kafka has the message
        // Otherwise we lose events on consumer crash
        pglogrepl.SendStandbyStatusUpdate(ctx, conn,
            pglogrepl.StandbyStatusUpdate{WALWritePosition: change.LSN})
    }
}

// Or skip all this and run Debezium — it does it for you.
# Consume Debezium-formatted CDC events from Kafka.
import json
from aiokafka import AIOKafkaConsumer

async def stream_orders_changes():
    consumer = AIOKafkaConsumer(
        "db.public.orders",                       # Debezium topic naming
        bootstrap_servers="kafka:9092",
        group_id="search-indexer",
        enable_auto_commit=False,                  # commit AFTER we apply
    )
    await consumer.start()
    try:
        async for msg in consumer:
            event = json.loads(msg.value)
            op = event["payload"]["op"]                  # c, u, d, r
            after = event["payload"].get("after")
            before = event["payload"].get("before")

            if op in ("c", "u", "r"):
                await search.index(after)
            elif op == "d":
                await search.delete(before["id"])

            await consumer.commit()                      # at-least-once
    finally:
        await consumer.stop()

    private final ObjectMapper json;

    @KafkaListener(topics = "db.public.orders",
                    groupId = "search-indexer",
                    containerFactory = "manualAckFactory")
    public void consume(ConsumerRecord<String, String> rec,
                        Acknowledgment ack) throws IOException {

        JsonNode root = json.readTree(rec.value());
        JsonNode payload = root.get("payload");
        String op = payload.get("op").asText();

        switch (op) {
            case "c", "u", "r" ->
                search.index(json.treeToValue(
                    payload.get("after"), Order.class));
            case "d" ->
                search.delete(payload.get("before").get("id").asLong());
        }

        ack.acknowledge();                              // commit only after success
    }
}

Polyglot persistence · the right tool

Core Architecture
Difficulty

The story

Twenty years ago, "the database" meant Oracle. Now there's a database for every workload, and the temptation is to use too many. The senior move is the opposite: start with Postgres, push it hard, and only add new datastores when you can name the specific bottleneck a new tool solves.

Real reasons to add: write throughput Postgres can't sustain (Cassandra/Scylla), full-text search Postgres can't answer well (Elasticsearch), analytical queries that take minutes (ClickHouse, Druid, BigQuery), or working set that exceeds RAM (Redis cache). Bad reasons: "everyone uses MongoDB now", "Mongo seems easier", "GraphDB sounds cool". Every datastore you add doubles the operational work, the dual-write surface area, and the on-call burden.

Add when
  • You've measured the bottleneck on the current store
  • The new tool solves it 10× — not 2×
  • You can operate it (or pay someone who does)
Avoid when
  • Choosing tools by trend, not by workload
  • Without a CDC story to keep stores in sync
  • Adding a 5th datastore to a 4-engineer team
Connects to CDC, Dual writes, Caching

# One service, three stores, one source of truth.
class UserService:
    def __init__(self, pg, redis, es):
        self.pg    = pg     # source of truth — Postgres
        self.redis = redis  # cache for hot reads
        self.es    = es     # Elasticsearch for full-text

    async def get_by_id(self, user_id: int) -> User | None:
        # Cache-aside: hit Redis first
        if cached := await self.redis.get(f"u:{user_id}"):
            return User.parse_raw(cached)
        user = await self.pg.fetchrow(
            "SELECT * FROM users WHERE id = $1", user_id)
        if user:
            await self.redis.setex(f"u:{user_id}", 300, User(**user).json())
        return user

    async def search(self, q: str) -> list[User]:
        # Postgres is bad at fuzzy text — use ES
        res = await self.es.search(
            index="users",
            query={"multi_match": {"query": q,
                                    "fields": ["name^2", "email"]}})
        return [User(**h["_source"]) for h in res["hits"]["hits"]]

    async def save(self, user: User):
        # Write ONLY to source of truth. CDC propagates to Redis + ES.
        await self.pg.execute("INSERT INTO users ...", user)

@Service
public class UserService {

    private final UserRepo          pg;       // system of record
    private final StringRedisTemplate redis;   // hot read cache
    private final ElasticsearchOps   es;      // full-text
    private final ObjectMapper      json;

    public User getById(Long id) throws IOException {
        String cached = redis.opsForValue().get("u:" + id);
        if (cached != null) return json.readValue(cached, User.class);

        User u = pg.findById(id).orElse(null);
        if (u != null) {
            redis.opsForValue().set("u:" + id,
                json.writeValueAsString(u), Duration.ofMinutes(5));
        }
        return u;
    }

    public List<User> search(String q) {
        Query query = NativeQuery.builder()
            .withQuery(qb -> qb.multiMatch(m -> m.query(q)
                .fields("name^2", "email"))).build();
        return es.search(query, User.class).stream()
            .map(SearchHit::getContent).toList();
    }

    @Transactional
    public void save(User u) {
        pg.save(u);    // CDC fans out to Redis + ES asynchronously
    }
}

The dual-write problem

Bug Bonus
Difficulty

The story

You write the order to Postgres. You publish an event to Kafka. Both succeed → all good. But what if the process crashes between the two writes? The DB has the order, Kafka doesn't, and now your warehouse, search index, and analytics are silently wrong forever. This is the dual-write problem, and it's the root cause of more silent data drift than any other bug.

You can't fix it with retries — the crash might happen after the DB commit but before any retry runs. You can't fix it with two-phase commit — it's slow, fragile, and most message brokers don't support it. The fix is structural: write once, in one transaction. The Outbox pattern stages the event in an outbox table inside the same DB transaction, then a separate publisher reads the table and publishes to Kafka. CDC accomplishes the same thing without an outbox table — Debezium tails the WAL.

Use when
  • Any flow that writes to a DB AND publishes an event
  • Sync between primary store and search/cache/analytics
  • Multi-step workflows that span systems
Avoid the trap of
  • Wrapping both writes in a try/catch and calling it done
  • Hoping retries will fix occasional drift
  • Ignoring it because "it almost never happens"
Connects to Outbox pattern, CDC, Exactly-once
// ✗ BROKEN — classic dual write. Crash between calls = data drift.
func placeOrderBroken(db *sql.DB, k *kafka.Writer, o Order) error {
    if _, err := db.Exec("INSERT INTO orders ..."); err != nil {
        return err
    }
    // CRASH HERE → DB has order, Kafka does not. Forever.
    return k.WriteMessages(ctx, kafka.Message{Value: marshal(o)})
}

// ✓ FIXED — outbox: data + event in ONE transaction
func placeOrder(db *sql.DB, o Order) error {
    tx, _ := db.Begin()
    defer tx.Rollback()

    if _, err := tx.Exec("INSERT INTO orders ...", o.ID); err != nil {
        return err
    }
    // Same transaction — atomic with the order
    payload, _ := json.Marshal(o)
    if _, err := tx.Exec(`
        INSERT INTO outbox(aggregate_id, type, payload, created_at)
        VALUES ($1, 'order.placed', $2, now())`,
        o.ID, payload); err != nil {
        return err
    }
    return tx.Commit()
    // A separate worker polls outbox + publishes to Kafka. At-least-once.
    // Or: Debezium reads the WAL directly. Either way — no dual write.
}
# ✗ BROKEN — two systems, no atomicity
async def place_order_broken(order):
    await db.execute("INSERT INTO orders ...", order)

# ✓ FIXED — outbox table, one transaction, async publisher
async def place_order(order: Order):
    async with db.transaction():
        await db.execute(
            "INSERT INTO orders(id, total) VALUES ($1, $2)",
            order.id, order.total)
        await db.execute("""
            INSERT INTO outbox(aggregate_id, type, payload)
            VALUES ($1, 'order.placed', $2)
        """, order.id, order.json())
    # Commit is atomic. Either both rows land, or neither.

# Separate worker — runs continuously, drains the outbox
async def drain_outbox():
    while True:
        events = await db.fetch("""
            SELECT id, type, payload FROM outbox
             WHERE published_at IS NULL
             ORDER BY id LIMIT 100
        """)
        for e in events:
            await kafka.send(e["type"], e["payload"])
            await db.execute(
                "UPDATE outbox SET published_at = now() WHERE id = $1",
                e["id"])
        await asyncio.sleep(0.5)

    private final KafkaTemplate<String, String> kafka;

    @Scheduled(fixedDelay = 500)
    public void drain() {
        outbox.findUnpublished(100).forEach(e -> {
            kafka.send(e.type(), e.payload())
                 .whenComplete((res, err) -> {
                     if (err == null) outbox.markPublished(e.id());
                 });
        });
    }
