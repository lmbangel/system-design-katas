01

Saga Pattern

Distributed Transactions Consistency Compensation
Difficulty

The story

A Saga breaks one big distributed transaction into a sequence of local transactions, each with a corresponding compensating action. If step 4 of 5 fails, you don't roll back globally — you run the inverses of steps 1, 2, and 3 to undo the world.

Two flavors: orchestrated (a central coordinator tells each service what to do next) and choreographed (services react to each other's events). Orchestration is easier to reason about; choreography couples less.

Sagas trade ACID for eventual consistency — for a brief window, the system is in an inconsistent state. Design compensations to be idempotent and commutative, because they will run more than once.

Use when
  • A workflow spans 3+ services
  • You can't hold a distributed lock
  • Brief inconsistency is acceptable
Avoid when
  • You need strong consistency
  • Compensations are impossible (sent emails)
  • A single DB transaction would work
// Orchestrated saga: each step has an action and a compensation.
type Step struct {
    Name    string
    Do      func(ctx context.Context) error
    Undo    func(ctx context.Context) error
}

func RunSaga(ctx context.Context, steps []Step) error {
    completed := []Step{}
    for _, s := range steps {
        if err := s.Do(ctx); err != nil {
            log.Printf("step %s failed: %v — compensating", s.Name, err)
            for i := len(completed) - 1; i >= 0; i-- {
                if cerr := completed[i].Undo(ctx); cerr != nil {
                    // Persist for later retry — never silently lose a compensation.
                    log.Printf("compensation %s failed: %v", completed[i].Name, cerr)
                }
            }
            return err
        }
        completed = append(completed, s)
    }
    return nil
}

// Usage
err := RunSaga(ctx, []Step{
    {"create-order",   orderSvc.Create,  orderSvc.Cancel},
    {"charge-card",    paySvc.Charge,    paySvc.Refund},
    {"reserve-stock",  stockSvc.Hold,    stockSvc.Release},
})
from dataclasses import dataclass
from typing import Callable, Awaitable, List
from fastapi import FastAPI, HTTPException

app = FastAPI()

@dataclass
class Step:
    name: str
    do: Callable[[], Awaitable[None]]
    undo: Callable[[], Awaitable[None]]

async def run_saga(steps: List[Step]) -> None:
    completed: List[Step] = []
    try:
        for s in steps:
            await s.do()
            completed.append(s)
    except Exception as e:
        for s in reversed(completed):
            try:
                await s.undo()
            except Exception as ce:
                # Persist to dead-letter for retry. Never lose a compensation.
                log.error(f"compensation {s.name} failed: {ce}")
        raise HTTPException(500, f"saga failed at {s.name}: {e}")

@app.post("/checkout")
async def checkout(order_id: str):
    await run_saga([
        Step("create-order",  lambda: order_svc.create(order_id),
                              lambda: order_svc.cancel(order_id)),
        Step("charge-card",   lambda: pay_svc.charge(order_id),
                              lambda: pay_svc.refund(order_id)),
        Step("reserve-stock", lambda: stock_svc.hold(order_id),
                              lambda: stock_svc.release(order_id)),
    ])
    return {"status": "ok"}
// SagaStep.java
public interface SagaStep {
    String name();
    void execute();
    void compensate();
}

// SagaOrchestrator.java
@Service
public class SagaOrchestrator {
    private static final Logger log = LoggerFactory.getLogger(SagaOrchestrator.class);

    public void run(List<SagaStep> steps) {
        List<SagaStep> done = new ArrayList<>();
        try {
            for (SagaStep s : steps) {
                s.execute();
                done.add(s);
            }
        } catch (Exception e) {
            log.warn("saga failed, compensating", e);
            Collections.reverse(done);
            for (SagaStep s : done) {
                try { s.compensate(); }
                catch (Exception ce) {
                    log.error("compensation {} failed", s.name(), ce);
                    // enqueue for retry — durable store
                }
            }
            throw new SagaFailedException(e);
        }
    }
}

// Usage
@PostMapping("/checkout")
public ResponseEntity<?> checkout(@RequestBody CheckoutDto dto) {
    orchestrator.run(List.of(
        new CreateOrderStep(dto),
        new ChargeCardStep(dto),
        new ReserveStockStep(dto)
    ));
    return ResponseEntity.ok().build();
}
02

Circuit Breaker

Resilience Fault Isolation Stateful
Difficulty

The story

A cascading failure happens when a slow downstream service holds connections open until your upstream service runs out of threads, then its upstream runs out, and so on. The Circuit Breaker is the electrical fuse: when the failure rate hits a threshold, it trips open and rejects calls instantly with a fallback — protecting both the caller and the dying downstream from a thundering herd.

Three states: Closed (normal), Open (rejecting calls), Half-Open (let one trial request through to probe recovery). Tune the failure threshold, the open duration, and the trial size carefully — too sensitive and you flap, too forgiving and you defeat the purpose.

Use when
  • Calling unreliable downstream services
  • Latency spikes are common
  • You have a sensible fallback
Avoid when
  • Calls are local and synchronous
  • Failures are user-input errors
  • You have no fallback strategy
// Minimal circuit breaker. In production prefer sony/gobreaker.
type State int
const ( Closed State = iota; Open; HalfOpen )

type Breaker struct {
    mu        sync.Mutex
    state     State
    failures  int
    threshold int
    openedAt  time.Time
    cooldown  time.Duration
}

func (b *Breaker) Call(fn func() error) error {
    b.mu.Lock()
    if b.state == Open {
        if time.Since(b.openedAt) < b.cooldown {
            b.mu.Unlock()
            return errors.New("circuit open")
        }
        b.state = HalfOpen  // allow a single trial
    }
    b.mu.Unlock()

    err := fn()

    b.mu.Lock()
    defer b.mu.Unlock()
    if err != nil {
        b.failures++
        if b.failures >= b.threshold || b.state == HalfOpen {
            b.state = Open
            b.openedAt = time.Now()
        }
        return err
    }
    b.failures = 0
    b.state = Closed
    return nil
}
import time, asyncio
from enum import Enum
from fastapi import FastAPI, HTTPException

class State(Enum):
    CLOSED, OPEN, HALF = "closed", "open", "half"

class Breaker:
    def __init__(self, threshold=5, cooldown=10.0):
        self.state = State.CLOSED
        self.failures, self.opened_at = 0, 0.0
        self.threshold, self.cooldown = threshold, cooldown
        self._lock = asyncio.Lock()

    async def call(self, fn):
        async with self._lock:
            if self.state == State.OPEN:
                if time.monotonic() - self.opened_at < self.cooldown:
                    raise RuntimeError("circuit open")
                self.state = State.HALF
        try:
            result = await fn()
        except Exception:
            async with self._lock:
                self.failures += 1
                if self.failures >= self.threshold or self.state == State.HALF:
                    self.state = State.OPEN
                    self.opened_at = time.monotonic()
            raise
        async with self._lock:
            self.failures, self.state = 0, State.CLOSED
        return result

app = FastAPI()
breaker = Breaker()

@app.get("/quote/{sym}")
async def quote(sym: str):
    try:
        return await breaker.call(lambda: prices.fetch(sym))
    except RuntimeError:
        return {"sym": sym, "price": None, "cached": True}  # fallback

}

            .retrieve().body(Quote.class);
    }

    // Same signature + Throwable. Called when breaker is open or call fails.

        return cache.getOrDefault(symbol, Quote.unknown(symbol));
    }

03

Retry with Backoff

Resilience Transient Failures Jitter
Difficulty

The story

Networks blink. Pods restart. A 503 right now might succeed in 200ms. Retrying handles these transient failures invisibly — but naive retries are dangerous: every client retrying at the same instant creates a synchronized stampede that can crater an already-struggling service.

The fix is two-fold: exponential backoff (1s, 2s, 4s, 8s…) doubles the wait each attempt, and jitter randomizes within each window so retries spread across the cooldown rather than piling up at the same millisecond. Always cap the maximum delay and the total attempt count.

Critical: only retry idempotent operations, or pair this with the Idempotency pattern. Retrying a non-idempotent POST charges the card twice.

Use when
  • Failures are transient (network, throttling)
  • The operation is idempotent
  • Latency budget allows extra attempts
Avoid when
  • The error is 4xx (client's fault)
  • The operation has side effects
  • Real-time, low-latency requirements
func RetryWithBackoff(ctx context.Context, maxAttempts int, fn func() error) error {
    base := 100 * time.Millisecond
    capDelay := 10 * time.Second

    var err error
    for attempt := 0; attempt < maxAttempts; attempt++ {
        if err = fn(); err == nil {
            return nil
        }
        if !isTransient(err) {
            return err  // don't retry 4xx
        }

        // exp backoff with full jitter (AWS recommended)
        delay := time.Duration(1<<attempt) * base
        if delay > capDelay { delay = capDelay }
        jittered := time.Duration(rand.Int63n(int64(delay)))

        select {
        case <-ctx.Done():
            return ctx.Err()
        case <-time.After(jittered):
        }
    }
    return fmt.Errorf("after %d attempts: %w", maxAttempts, err)
}

// usage
err := RetryWithBackoff(ctx, 5, func() error {
    return httpClient.Post("/api/charge", payload)
})

from tenacity import (retry, stop_after_attempt, wait_exponential_jitter,
                      retry_if_exception_type)
import httpx
from fastapi import FastAPI

app = FastAPI()

# tenacity gives you battle-tested retry logic for free
@retry(
    stop=stop_after_attempt(5),
    wait=wait_exponential_jitter(initial=0.1, max=10),
    retry=retry_if_exception_type((httpx.TransportError, httpx.HTTPStatusError)),

async def fetch_quote(symbol: str) -> dict:
    async with httpx.AsyncClient(timeout=2.0) as c:
        r = await c.get(f"https://prices/api/{symbol}")
        if 500 <= r.status_code < 600:
            r.raise_for_status()        # will trigger retry
        return r.json()

# Manual version (no library) — same shape
async def retry_manual(fn, attempts=5, base=0.1, cap=10):
    for i in range(attempts):
        try:
            return await fn()
        except Exception:
            if i == attempts - 1: raise
            delay = min(cap, base * 2 ** i)
            await asyncio.sleep(random.uniform(0, delay))   # full jitter

@app.get("/quote/{sym}")
async def quote(sym: str):
    return await fetch_quote(sym)

@Configuration
public class RetryConfig {
    @Bean
    public RetryRegistry retryRegistry() {
        RetryConfig cfg = RetryConfig.custom()
            .maxAttempts(5)
            .intervalFunction(
                IntervalFunction.ofExponentialRandomBackoff(
                    Duration.ofMillis(100), 2.0, 0.5)) // jitter factor
            .retryOnException(e ->
                e instanceof IOException
                || (e instanceof HttpServerErrorException))
            .build();
        return RetryRegistry.of(cfg);
    }
}

04

API Gateway

Edge Routing Cross-Cutting
Difficulty

The story

If every client knows the address of every service, every change ripples outward. The API Gateway is a single ingress that fronts your whole microservice estate. It handles cross-cutting concerns — authentication, rate limiting, TLS termination, request logging, caching, retries — so each service doesn't have to.

It also hides internal topology. A mobile client makes one call to /orders/123/full and the gateway fans out to four services and assembles the response. This is the Backend-for-Frontend (BFF) variant: a tailored gateway per client type.

Watch for the trap: gateways accumulate business logic until they become a monolith disguised as infrastructure. Keep them thin — routing and policy, not domain rules.

Use when
  • You have 3+ services exposed externally
  • Multiple client types (web, mobile, partners)
  • Centralized auth and rate limits matter
Avoid when
  • Single service, single client
  • You'd put domain logic in the gateway
  • Latency budget can't afford an extra hop
// Tiny reverse-proxy gateway with auth + rate limit middleware
type Route struct {
    Prefix string
    Target *url.URL
}

func NewGateway(routes []Route) http.Handler {
    mux := http.NewServeMux()
    for _, r := range routes {
        proxy := httputil.NewSingleHostReverseProxy(r.Target)
        mux.Handle(r.Prefix, http.StripPrefix(r.Prefix, proxy))
    }
    return authMiddleware(rateLimitMiddleware(mux))
}

func authMiddleware(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        tok := r.Header.Get("Authorization")
        claims, err := verifyJWT(tok)
        if err != nil {
            http.Error(w, "unauthorized", 401); return
        }
        r.Header.Set("X-User-ID", claims.Subject)  // pass downstream
        next.ServeHTTP(w, r)
    })
}

func main() {
    u, _ := url.Parse
    orderURL, _ := url.Parse("http://orders:8080")
    userURL,  _ := url.Parse("http://users:8080")

    h := NewGateway([]Route{
        {"/orders/", orderURL},
        {"/users/",  userURL},
    })
    http.ListenAndServe(":8000", h)
}

app = FastAPI(title="API Gateway")
client = httpx.AsyncClient(timeout=5.0)

ROUTES = {
    "orders":  "http://orders:8080",
    "users":   "http://users:8080",
    "catalog": "http://catalog:8080",

def require_auth(request: Request) -> dict:
    auth = request.headers.get("authorization", "")
    if not auth.startswith("Bearer "):
        raise HTTPException(401)
    try:
        return jwt.decode(auth[7:], SECRET, algorithms=["HS256"])
    except jwt.PyJWTError:
        raise HTTPException(401)

@app.api_route("/{service}/{path:path}",
               methods=["GET", "POST", "PUT", "DELETE", "PATCH"])
async def proxy(service: str, path: str, request: Request,
                user: dict = Depends(require_auth)):
    target = ROUTES.get(service)
    if not target:
        raise HTTPException(404)
    headers = {**dict(request.headers), "x-user-id": user["sub"]}
    headers.pop("host", None)
    r = await client.request(
        request.method, f"{target}/{path}",
        headers=headers,
        params=request.query_params,
        content=await request.body(),
    )
    return Response(r.content, status_code=r.status_code,
                    headers=dict(r.headers))

05

Idempotency

Correctness Safety Dedup
Difficulty

The story

The network can fail after the server processed the request but before the response arrives. The client retries. Now the order is placed twice. The card is charged twice. The user sends two angry tweets.

An idempotent operation can be performed multiple times with the same effect as performing it once. The standard implementation: clients pass an Idempotency-Key header (a UUID per logical attempt). The server stores (key → response) the first time it processes the request, and on subsequent calls with the same key, replays the cached response without re-running the work.

GET, PUT, and DELETE are naturally idempotent. POST is the dangerous one — that's where you need an explicit key. Stripe popularized this pattern; it's now a baseline expectation for any payments-adjacent API.

Use when
  • Operations have side effects (charges, sends)
  • Clients retry on timeout
  • You can't tolerate duplicate work
Avoid when
  • Operation is already idempotent (PUT)
  • Read-only endpoints
  • Cost of dedup store outweighs duplicate risk
// Idempotency middleware — stores request hash + response in Redis
type CachedResp struct {
    Status  int
    Body    []byte
    ReqHash string  // guards key reuse with a different payload
}

func Idempotent(rdb *redis.Client, ttl time.Duration) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            key := r.Header.Get("Idempotency-Key")
            if key == "" || r.Method != http.MethodPost {
                next.ServeHTTP(w, r); return
            }
            body, _ := io.ReadAll(r.Body)
            r.Body = io.NopCloser(bytes.NewReader(body))
            hash := fmt.Sprintf("%x", sha256.Sum256(body))

            // Try to fetch cached response
            if raw, err := rdb.Get(r.Context(), "idem:"+key).Bytes(); err == nil {
                var cr CachedResp
                json.Unmarshal(raw, &cr)
                if cr.ReqHash != hash {
                    http.Error(w, "key reused with different body", 422); return
                }
                w.WriteHeader(cr.Status); w.Write(cr.Body); return
            }

            // Capture response, then store
            rec := &recorder{ResponseWriter: w, buf: &bytes.Buffer{}, status: 200}
            next.ServeHTTP(rec, r)
            payload, _ := json.Marshal(CachedResp{rec.status, rec.buf.Bytes(), hash})
            rdb.Set(r.Context(), "idem:"+key, payload, ttl)
        })
    }
}
from fastapi import FastAPI, Request, Header, HTTPException
from fastapi.responses import JSONResponse
import hashlib, json, redis.asyncio as redis

app = FastAPI()
rdb = redis.Redis(host="redis")
TTL = 86400  # 24h

@app.middleware("http")
async def idempotency(request: Request, call_next):
    key = request.headers.get("idempotency-key")
    if not key or request.method != "POST":
        return await call_next(request)

    body = await request.body()
    body_hash = hashlib.sha256(body).hexdigest()
    cache_key = f"idem:{key}"

    cached = await rdb.get(cache_key)
    if cached:
        data = json.loads(cached)
        if data["hash"] != body_hash:
            raise HTTPException(422, "key reused with different body")
        return JSONResponse(data["body"], status_code=data["status"])

    # Use SET NX as a lock so concurrent retries don't both run the work
    locked = await rdb.set(f"lock:{key}", "1", nx=True, ex=30)
    if not locked:
        raise HTTPException(409, "request in flight")

    response = await call_next(request)
    resp_body = b"".join([c async for c in response.body_iterator])
    await rdb.setex(cache_key, TTL, json.dumps({
        "hash": body_hash,
        "status": response.status_code,
        "body": json.loads(resp_body or b"null"),
    }))
    return JSONResponse(json.loads(resp_body), status_code=response.status_code)
@Component
public class IdempotencyFilter implements Filter {

    private final StringRedisTemplate redis;
    private final ObjectMapper mapper;
    private static final Duration TTL = Duration.ofHours(24);

    @Override
    public void doFilter(ServletRequest req, ServletResponse res, FilterChain chain)
            throws IOException, ServletException {

        HttpServletRequest  httpReq = (HttpServletRequest) req;
        HttpServletResponse httpRes = (HttpServletResponse) res;

        String key = httpReq.getHeader("Idempotency-Key");
        if (key == null || !"POST".equalsIgnoreCase(httpReq.getMethod())) {
            chain.doFilter(req, res); return;
        }

        CachedRequestWrapper wrapped = new CachedRequestWrapper(httpReq);
        String bodyHash = DigestUtils.sha256Hex(wrapped.getCachedBody());

        String cached = redis.opsForValue().get("idem:" + key);
        if (cached != null) {
            CachedResponse cr = mapper.readValue(cached, CachedResponse.class);
            if (!cr.hash().equals(bodyHash)) {
                httpRes.sendError(422, "key reused with different body"); return;
            }
            httpRes.setStatus(cr.status());
            httpRes.getWriter().write(cr.body());
            return;
        }

        ContentCachingResponseWrapper resWrap =
            new ContentCachingResponseWrapper(httpRes);
        chain.doFilter(wrapped, resWrap);

        String body = new String(resWrap.getContentAsByteArray());
        redis.opsForValue().set("idem:" + key,
            mapper.writeValueAsString(
                new CachedResponse(bodyHash, resWrap.getStatus(), body)),
            TTL);
        resWrap.copyBodyToResponse();
    }
}
06

Event-Driven Architecture

Async Decoupling Pub/Sub
Difficulty

The story

In synchronous architectures, the order service calls inventory, then notifications, then analytics — and waits for each. If notifications is slow, the user's checkout is slow. If analytics is down, the order fails. Tight coupling hides everywhere.

Event-driven architecture inverts the flow: the order service emits one event — OrderPlaced — to a broker (Kafka, NATS, RabbitMQ) and goes home. Anyone interested subscribes. Inventory reacts. Notifications reacts. Analytics reacts. The producer doesn't know they exist, and adding a fifth consumer requires zero changes upstream.

Trade-offs: you gain scalability, loose coupling, audit trails. You lose simple debugging — flows are now distributed across services and time. Use a correlation ID on every event and invest in tracing (OpenTelemetry) from day one.

Use when
  • Multiple services react to the same fact
  • You want loose coupling and scaling
  • Async processing is acceptable
Avoid when
  • You need a synchronous answer
  • Strong ordering across consumers required
  • Team isn't ready for async debugging
// Producer
type OrderPlaced struct {
    EventID      string    `json:"event_id"`
    OccurredAt   time.Time `json:"occurred_at"`
    OrderID      string    `json:"order_id"`
    UserID       string    `json:"user_id"`
    TotalCents   int64     `json:"total_cents"`
}

func PublishOrderPlaced(w *kafka.Writer, e OrderPlaced) error {
    payload, _ := json.Marshal(e)
    return w.WriteMessages(context.Background(), kafka.Message{
        Topic: "orders.placed",
        Key:   []byte(e.OrderID),  // partition key — same order ⇒ same partition
        Value: payload,
        Headers: []kafka.Header{

Bulkhead

Isolation Resource Limits Containment
Difficulty

The story

Named after the watertight compartments in a ship's hull: if one floods, the others stay dry and the ship floats. In software, the Bulkhead isolates resources — connection pools, thread pools, semaphores — so a single misbehaving downstream can't drain the whole process.

Without bulkheads, a slow third-party API can hold every HTTP client thread in your service, blocking calls to healthy dependencies and turning a partial outage into a total one. With bulkheads, each downstream gets its own bounded resource pool. When pool B is saturated, calls to B fail fast — A and C remain responsive.

Often paired with the Circuit Breaker: bulkheads contain saturation, breakers detect it. Together they prevent the most insidious failure mode in distributed systems — cascading resource exhaustion.

Use when
  • One service has multiple downstreams
  • Critical and non-critical paths share infra
  • You see resource exhaustion in incidents
Avoid when
  • Single downstream dependency
  • Resource overhead exceeds the benefit
  • Traffic is uniformly small
// One bounded HTTP client per downstream — separate connection pools.
// Plus a semaphore to cap in-flight calls per dependency.
type Bulkhead struct {
    name    string
    client  *http.Client
    sem     chan struct{}
}

func NewBulkhead(name string, maxConcurrent int) *Bulkhead {
    return &Bulkhead{
        name: name,
        client: &http.Client{
            Timeout: 3 * time.Second,
            Transport: &http.Transport{
                MaxIdleConns:        50,
                MaxConnsPerHost:     maxConcurrent,  // hard cap
                MaxIdleConnsPerHost: 20,
                IdleConnTimeout:     90 * time.Second,
            },
        },
        sem: make(chan struct{}, maxConcurrent),
    }
}

func (b *Bulkhead) Do(ctx context.Context, req *http.Request) (*http.Response, error) {
    select {
    case b.sem <- struct{}{}:
        defer func() { <-b.sem }()
    case <-ctx.Done():
        return nil, ctx.Err()
    default:
        return nil, fmt.Errorf("bulkhead %s full · fail-fast", b.name)
    }
    return b.client.Do(req.WithContext(ctx))
}

// Each downstream gets its own bulkhead
var (
    paymentsBH  = NewBulkhead("payments",  20)
    analyticsBH = NewBulkhead("analytics", 5)   // non-critical, smaller pool
)
import asyncio, httpx
from fastapi import FastAPI, HTTPException

class Bulkhead:
    def __init__(self, name: str, base_url: str, max_concurrent: int):
        self.name = name
        self.sem = asyncio.Semaphore(max_concurrent)
        self.client = httpx.AsyncClient(
            base_url=base_url,
            timeout=httpx.Timeout(3.0),
            limits=httpx.Limits(
                max_connections=max_concurrent,    # hard pool cap
                max_keepalive_connections=max_concurrent // 2,
            ),
        )

    async def request(self, method: str, path: str, **kw) -> httpx.Response:
        if self.sem.locked() and self.sem._value == 0:
            raise HTTPException(503, f"bulkhead {self.name} full")
        async with self.sem:
            return await self.client.request(method, path, **kw)

Service Discovery

Infrastructure Networking
Difficulty

The story

In a static world you hard-code orders.example.com:8080 and forget about it. In containers and autoscaling groups, that machine doesn't exist tomorrow. Service Discovery is a dynamic phone book: each instance registers itself on startup, sends heartbeats while alive, and is evicted when it dies. Clients ask the registry "where is orders?" and get back a healthy instance — round-robined, weighted, or zone-aware.

Two flavors. Client-side (Eureka, Consul SDK): the client gets the list and load-balances itself. Server-side (Kubernetes Service, AWS ALB): the platform hides discovery behind a stable virtual IP. Most teams now lean on the platform — kube-dns is service discovery you didn't have to write.

Use when
  • Instances scale up/down dynamically
  • You need health-aware routing
  • Multi-zone or multi-region traffic shaping
Avoid when
  • You're already on Kubernetes (use Services)
  • Two-service systems with stable hostnames
  • Team can't operate the registry's HA story

CQRS · Command Query Responsibility Segregation

Bonus Data Architecture
Difficulty

The story

Reads and writes have nothing in common. Writes need validation, transactions, and a normalized schema. Reads need fast denormalized projections joined for the screen they serve. Forcing both through one model means slow queries on the read side and accidentally complex aggregates on the write side. CQRS separates them: the command side mutates the source of truth; the query side serves precomputed read models built from events emitted by the command side.

It pairs naturally with event sourcing but doesn't require it. The cost is real: two models, eventual consistency, and "I just placed an order but I can't see it yet" UX problems. Don't reach for it on day one. Reach for it when a single read request joins seven tables across three services.

Use when
  • Read and write workloads scale very differently
  • Queries need denormalized views across services
  • You're already event-driven
Avoid when
  • CRUD app with simple read patterns
  • Strong read-your-writes is non-negotiable
  • Team isn't ready to operate two stores
// Two handlers, two stores. Same domain, different shapes.

// --- Command side: writes to Postgres, emits event ---
type CreateOrder struct {
    CustomerID string
    Items      []Item
}

func (h *CmdHandler) Handle(cmd CreateOrder) (string, error) {
    id := uuid.NewString()

    if err != nil { tx.Rollback(); return "", err }
    tx.Commit()

}

// --- Query side: reads denormalized view from Redis/Mongo ---

    ID           string
    CustomerName string   // already joined
    Total        float64  // already computed
    ItemSummary  string
}

func (h *QueryHandler) GetOrder(id string) (*OrderView, error) {
    return h.readStore.FindOne(id)   // single round-trip, no joins
}

func (p *Projector) OnOrderCreated(e OrderCreated) {
    cust := p.customers.Get(e.Customer)
    p.readStore.Upsert(OrderView{ID: e.ID, CustomerName: cust.Name})
}

Transactional Outbox

Bonus Reliability Data
Difficulty

The story

The classic dual-write problem: you save the order to Postgres, then publish "order.created" to Kafka. Between those two calls, your process can crash. Now your DB and your event log disagree forever — silent data corruption that surfaces as a customer-support ticket three weeks later.

The Outbox trick: in the same DB transaction that writes the business row, also insert a row into an outbox table. Either both land or neither does — that's what ACID gives you. A separate relay process polls the outbox (or tails the WAL via CDC tools like Debezium), publishes to the broker, and marks the row sent. You now have at-least-once delivery from a transactional source — pair with idempotent consumers and you're golden.

Use when
  • You write to a DB and emit events from the same operation
  • You can't lose events even if the broker is down
  • You want exactly-once-effects without 2PC
Avoid when
  • You don't actually publish events
  • Best-effort fire-and-forget is acceptable
  • Your DB doesn't support transactions
// Same TX writes business row AND outbox row. Relay publishes later.
func CreateOrder(ctx context.Context, db *sql.DB, o Order) error {
    tx, err := db.BeginTx(ctx, nil)
    if err != nil { return err }
    defer tx.Rollback()

    // 1. Business row
    if _, err := tx.ExecContext(ctx,
        "INSERT INTO orders(id, customer_id, total) VALUES($1,$2,$3)",
        o.ID, o.CustomerID, o.Total); err != nil {
        return err
    }

    // 2. Outbox row — same transaction, atomic with above
    payload, _ := json.Marshal(OrderCreated{ID: o.ID, Total: o.Total})
    if _, err := tx.ExecContext(ctx,
        `INSERT INTO outbox(id, topic, payload, created_at)
         VALUES($1, $2, $3, NOW())`,
        uuid.NewString(), "order.created", payload); err != nil {
        return err
    }

    return tx.Commit()  // either both committed, or neither
}

// Separate goroutine — the Relay
func RelayLoop(ctx context.Context, db *sql.DB, producer *kafka.Writer) {
    ticker := time.NewTicker(500 * time.Millisecond)
    for range ticker.C {
        rows, _ := db.QueryContext(ctx,
            `SELECT id, topic, payload FROM outbox
             WHERE sent_at IS NULL ORDER BY created_at LIMIT 100
             FOR UPDATE SKIP LOCKED`)
        for rows.Next() {
            var id, topic string; var payload []byte
            rows.Scan(&id, &topic, &payload)
            producer.WriteMessages(ctx, kafka.Message{Topic: topic, Value: payload})
            db.ExecContext(ctx, "UPDATE outbox SET sent_at=NOW() WHERE id=$1", id)
        }
    }
}

# --- Relay: separate worker, pulled and published ---
async def relay_loop():
    while True:
        async with pg.transaction():
            rows = await pg.fetch(
                """SELECT id, topic, payload FROM outbox
                          WHERE sent_at IS NULL
                          ORDER BY created_at LIMIT 100
                          FOR UPDATE SKIP LOCKED"""
            )
            for r in rows:
                await producer.send_and_wait(
                    r["topic"], r["payload"].encode()
                )
                await pg.execute(
                    "UPDATE outbox SET sent_at=NOW() WHERE id=$1", r["id"]
                )
        await asyncio.sleep(0.5)

Sidecar

Bonus Infrastructure Networking
Difficulty

The story

Every service needs the same boring stuff: mTLS, retries, circuit breakers, distributed tracing, auth, rate limits. If you build it inside each service, you have N implementations drifting in N languages. The Sidecar says: run a companion process next to the app — same pod, same lifecycle — and let it handle all the cross-cutting concerns. The app speaks plain http://localhost; the sidecar handles the rest.

This is the foundation of every service mesh — Istio, Linkerd, Consul Connect — and the SDKs sitting next to your serverless functions. Trade-off: you've added a hop, a process, and a configuration surface. For most teams running 10+ services on Kubernetes, that's a price worth paying. For three services on Heroku, it's overkill.

Use when
  • You run polyglot services and want one networking story
  • You need mTLS, mesh-wide tracing, or fine-grained policy
  • You're already on Kubernetes
Avoid when
  • Small fleet, one language, simple network
  • Latency budget is razor-thin
  • Team can't operate a mesh control plane

    private final RestClient client = RestClient.create();

    @PostMapping("/checkout")
    public ResponseEntity<String> checkout(@RequestBody Cart cart) {
        // No TLS context. No retry interceptor. No tracing setup.
        // The Envoy sidecar in this pod handles all of it.
        String body = client.post()
            .uri("http://payments/charge")
            .body(cart)
            .retrieve()
            .body(String.class);
        return ResponseEntity.ok(body);
    }
}

12

Strangler Fig

Bonus Migration Architecture
Difficulty

The story

Named after the strangler fig — a vine that grows up an old tree until the host eventually dies inside it. Same idea for legacy code: instead of a Big Rewrite that ships in eighteen months and breaks everything, you put a facade in front of the monolith and start extracting one capability at a time. /orders/* goes to a new service. Everything else still hits the legacy. Next quarter, /inventory/* moves. Then /users/*. Eventually the monolith is a husk and you delete it.

The win is risk: every step is small, reversible, and individually shippable. The trap is starting without a facade, or extracting things piecemeal without a plan, and ending up with a monolith and a distributed system. Pick a clear seam, build the facade first, then move.

Use when
  • You have a legacy monolith you can't take down
  • You need to keep shipping features during migration
  • Stakeholders won't fund an 18-month rewrite
Avoid when
  • The system is small enough to rewrite in a sprint
  • You can't introduce a routing layer in front
  • You don't yet know your service seams
// Reverse-proxy facade — route extracted paths to the new service,
// fall through to the monolith for everything else.
package main

import (
    "net/http"
    "net/http/httputil"
    "net/url"
    "strings"
)

func main() {
    newOrders, _ := url.Parse("http://orders-svc:8080")
    monolith, _ := url.Parse("http://legacy:8080")

    orderProxy := httputil.NewSingleHostReverseProxy(newOrders)
    legacyProxy := httputil.NewSingleHostReverseProxy(monolith)

    // extracted paths — keep this list growing as you peel off capabilities
    extracted := []string{"/orders", "/api/v1/orders"}

    http.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
        for _, prefix := range extracted {
            if strings.HasPrefix(r.URL.Path, prefix) {
                orderProxy.ServeHTTP(w, r)
                return
            }
        }
        // Everything else still hits the legacy monolith
        legacyProxy.ServeHTTP(w, r)
    })

    http.ListenAndServe(":80", nil)
}

// Optional: shadow traffic to the new service first, compare responses,
// then flip the route once you trust parity.
# FastAPI facade — extracted routes go to the new service,
# everything else proxies to the legacy app.
from fastapi import FastAPI, Request
from fastapi.responses import Response
import httpx

app = FastAPI()
client = httpx.AsyncClient(timeout=10.0)

EXTRACTED = ("/orders", "/api/v1/orders")
NEW_SVC   = "http://orders-svc:8080"
LEGACY    = "http://legacy:8080"

@app.api_route("/{full_path:path}", methods=["GET", "POST", "PUT", "DELETE", "PATCH"])
async def facade(full_path: str, request: Request):
    target_root = NEW_SVC if _is_extracted(full_path) else LEGACY
    url = f"{target_root}/{full_path}"

    body = await request.body()
    headers = {k: v for k, v in request.headers.items() if k.lower() != "host"}
    r = await client.request(
        request.method, url,
        params=request.query_params,
        content=body, headers=headers,
    )
    return Response(content=r.content, status_code=r.status_code, headers=dict(r.headers))

def _is_extracted(path: str) -> bool:
    p = "/" + path.lstrip("/")
    return any(p.startswith(prefix) for prefix in EXTRACTED)

# As more capabilities move out, EXTRACTED grows. Eventually the legacy
# branch is dead code and the monolith can be retired.
// Spring Cloud Gateway — declarative routing for the strangler facade.

@Configuration
public class StranglerRoutes {

    @Bean
    public RouteLocator routes(RouteLocatorBuilder b) {
        return b.routes()
            // Extracted: orders capability now lives in its own service
            .route("orders-new", r -> r
                .path("/orders/**", "/api/v1/orders/**")
                .uri("http://orders-svc:8080"))

            // Extracted: inventory just moved last sprint
            .route("inventory-new", r -> r
                .path("/inventory/**")
                .uri("http://inventory-svc:8080"))

            // Catch-all — everything we haven't peeled off yet still goes to the monolith
            .route("legacy", r -> r
                .path("/**")
                .uri("http://legacy:8080"))
            .build();
    }
}

/*
  Migration roadmap:
   1. Stand up the gateway in front of the monolith — every route goes to legacy.
   2. Pick the most isolated capability. Extract its data and behavior.
   3. Add a route that points to the new service.
   4. Run shadow traffic, compare responses, build confidence.
   5. Flip the route. Delete the old code path inside the monolith.
   6. Repeat until the monolith is empty.
*/
